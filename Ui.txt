local function _decodeUrl(b)
    local s = {}
    for i = 1, #b do
        local a, k = b[i], 0x5A
        local r, p = 0, 1
        while a > 0 or k > 0 do
            local ra, rk = a % 2, k % 2
            if ra ~= rk then r = r + p end
            a, k, p = (a - ra) / 2, (k - rk) / 2, p * 2
        end
        s[i] = string.char(r)
    end
    return table.concat(s)
end

RegisterHit, RegisterAttack = nil, nil

function AttackMultipleTargets(targets)
    if not RegisterHit or not RegisterAttack then return end
    pcall(function()
        if not targets or #targets == 0 then return end
        local allTargets = {}
        for _, char in pairs(targets) do
            local head = char:FindFirstChild("Head")
            if head then table.insert(allTargets, {char, head}) end
        end
        if #allTargets == 0 then return end
        RegisterAttack:FireServer(0)
        RegisterHit:FireServer(allTargets[1][2], allTargets)
    end)
end

function StartFastAttack()
    if FastAttackRunning then return end
    FastAttackRunning = true
    spawn(function()
        while FastAttackEnabled do
            RunService.Stepped:Wait()
            local myChar = player.Character
            local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")
            if myHRP then
                local targets = {}
                for _, p in pairs(Players:GetPlayers()) do
                    if p ~= player and p.Character then
                        local hum = p.Character:FindFirstChild("Humanoid")
                        local hrp = p.Character:FindFirstChild("HumanoidRootPart")
                        if hum and hrp and hum.Health > 0 and (hrp.Position - myHRP.Position).Magnitude <= FastAttackRange then
                            table.insert(targets, p.Character)
                        end
                    end
                end
                local enemies = workspace:FindFirstChild("Enemies")
                if enemies then
                    for _, enemy in pairs(enemies:GetChildren()) do
                        local hum = enemy:FindFirstChild("Humanoid")
                        local hrp = enemy:FindFirstChild("HumanoidRootPart")
                        if hum and hrp and hum.Health > 0 and (hrp.Position - myHRP.Position).Magnitude <= FastAttackRange then
                            table.insert(targets, enemy)
                        end
                    end
                end
                if #targets > 0 then AttackMultipleTargets(targets) end
            end
        end
        FastAttackRunning = false
    end)
end

local function isHoldingPortalFruit()
    local char = player.Character
    if not char then return false end
    local tool = char:FindFirstChildOfClass("Tool")
    if tool then
        local tName = string.lower(tool.Name)
        if string.find(tName, "portal") or string.find(tName, "door") then return true end
    end
    return false
end

local function equipSanguineArt()
    local char = player.Character
    if not char then return nil end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return nil end
    
    local tool = char:FindFirstChild("Sanguine Art") or char:FindFirstChild("Sanguine")
    if not tool then
        local bp = player:FindFirstChild("Backpack")
        if bp then
            for _, t in ipairs(bp:GetChildren()) do
                local name = string.lower(t.Name)
                if string.find(name, "sanguine") then
                    hum:EquipTool(t)
                    tool = t
                    task.wait(0.1)
                    break
                end
            end
        end
    end
    return tool
end

end

FLASH_IDS = {"17555632156","18461649274","616006778","616010882","5403485593","5403491911"}

function isFlashstep(track)
    for _,n in ipairs(FLASH_NAMES) do
        if string.find(string.lower(track.Name), string.lower(n)) then return true end
    end
    local id = track.Animation and track.Animation.AnimationId:match("%d+") or ""
    for _,fid in ipairs(FLASH_IDS) do
        if id == fid then return true end
    end
    return false
end

function IsSilentAimEnemy(p)
    if not p or p == player then return false end
    local myTeam, targetTeam = player.Team, p.Team
    if myTeam and targetTeam and myTeam.Name == "Marines" and targetTeam.Name == "Marines" then
        return false
    end
    return true
end

function GetClosestTargetToMouse()
    local screenCenter = Vector2.new(camera.ViewportSize.X / 2, camera.ViewportSize.Y / 2)
    local closest, shortest = nil, (_G.G_SilentAimShowFOV and _G.G_SilentAimFOV or math.huge)

    if AimlockTargetPlayer and AimlockTargetPlayer ~= "Nearest" then
        local specificPlayer = Players:FindFirstChild(AimlockTargetPlayer)
        if specificPlayer and specificPlayer.Character and not BlacklistedPlayers[specificPlayer.Name] then
            local hum = specificPlayer.Character:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health > 0 then
                local part = specificPlayer.Character:FindFirstChild(_G.G_SilentAimPart) or specificPlayer.Character:FindFirstChild("HumanoidRootPart")
                if part then return part end
            end
        end
    end

    local function checkChar(char)
        if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health <= 0 then return end
        local part = char:FindFirstChild(_G.G_SilentAimPart) or char:FindFirstChild("HumanoidRootPart")
        if not part then return end
        local pos, onScreen = camera:WorldToViewportPoint(part.Position)
        if onScreen then
            local dist = (Vector2.new(pos.X, pos.Y) - screenCenter).Magnitude
            if dist < shortest then
                closest = part
                shortest = dist
            end
        end
    end

    if _G.G_SilentAimTargetPlayers then
        for _, p in pairs(Players:GetPlayers()) do
            if p ~= player and p.Character and not BlacklistedPlayers[p.Name] then
                if not _G.G_SilentAimTeamCheck or IsSilentAimEnemy(p) then
                    checkChar(p.Character)
                end
            end
        end
    end

    if _G.G_SilentAimTargetMobs then
        local enemies = workspace:FindFirstChild("Enemies")
        if enemies then
            for _, mob in pairs(enemies:GetChildren()) do
                checkChar(mob)
            end
        end
    end

    return closest
end

        local realStore = { Hit = rawget(MouseModule, "Hit"), Target = rawget(MouseModule, "Target") }
        local mmt = getrawmetatable(MouseModule)
        if mmt then setreadonly(mmt, false) else mmt = {}; setmetatable(MouseModule, mmt) end
        rawset(MouseModule, "Hit", nil); rawset(MouseModule, "Target", nil)
        mmt.__index = function(self, key)
            if key == "Hit" then
                if _G.G_SilentAimSkill and currentSilentAimTarget then return CFrame.new(currentSilentAimTarget.Position) end
                return realStore.Hit
            elseif key == "Target" then
                if _G.G_SilentAimSkill and currentSilentAimTarget then return currentSilentAimTarget end
                return realStore.Target
            end
        end
        mmt.__newindex = function(self, key, value)
            if key == "Hit" or key == "Target" then realStore[key] = value else rawset(self, key, value) end
        end
        setreadonly(mmt, true)
    end)
end

function ClearTracers()
    for _, line in pairs(TracersList) do
        pcall(function()
            if line.Remove then line:Remove() elseif line.Destroy then line:Destroy() end
        end)
    end
    TracersList = {}
end

function UpdateTracers()
    if not ESPMaster or not ESPTracers then
        ClearTracers()
        return
    end
    
    local screenCenter = Vector2.new(camera.ViewportSize.X / 2, camera.ViewportSize.Y - 10)
    local activePlayers = {}
    
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= player and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health > 0 then
                local hrpPos, onScreen = camera:WorldToViewportPoint(p.Character.HumanoidRootPart.Position)
                if onScreen then
                    table.insert(activePlayers, {Player = p, Pos = Vector2.new(hrpPos.X, hrpPos.Y)})
                end
            end
        end
    end

    if Drawing then
        for i, data in ipairs(activePlayers) do
            local line = TracersList[i]
            if not line then
                line = Drawing.new("Line")
                line.Thickness = 1.5
                line.Transparency = 0.8
                TracersList[i] = line
            end
            line.Color = currentThemeColor
            line.From = screenCenter
            line.To = data.Pos
            line.Visible = true
        end
        for i = #activePlayers + 1, #TracersList do
            if TracersList[i] then TracersList[i].Visible = false end
        end
    end
end

function UpdateESP()

        ClearTracers()
        return

        end

function UpdateESPData()
    if not ESPMaster then 
        ClearTracers()
        return 
    end
    local myChar = player.Character
    local myHrp = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myHrp then return end

    for _, billboard in pairs(ESPObjects) do
        if billboard and billboard.Parent then
            local char = billboard.Parent.Parent
            if char then
                local hum = char:FindFirstChildOfClass("Humanoid")
                local hrp = char:FindFirstChild("HumanoidRootPart")
                if hum and hrp then
                    local hpPct = hum.Health / math.max(hum.MaxHealth, 1)
                    local hBg = billboard:FindFirstChild("HealthBg")
                    if hBg then
                        local hFill = hBg:FindFirstChild("HealthFill")
                        if hFill then
                            hFill.Size = UDim2.new(math.clamp(hpPct, 0, 1), 0, 1, 0)
                            if hpPct > 0.6 then
                                hFill.BackgroundColor3 = Color3.fromRGB(0, 255, 120)
                            elseif hpPct > 0.3 then
                                hFill.BackgroundColor3 = Color3.fromRGB(255, 200, 0)
                            else
                                hFill.BackgroundColor3 = Color3.fromRGB(255, 50, 50)
                            end
                        end
                        hBg.Visible = ESPHealth
                    end
                    local dist = math.floor((hrp.Position - myHrp.Position).Magnitude)
                    local dText = billboard:FindFirstChild("DistLabel")
                    if dText then
                        dText.Text = "[" .. dist .. " studs]"
                        dText.Visible = ESPDist
                    end
                    local nText = billboard:FindFirstChild("NameLabel")
                    if nText then
                        nText.Visible = ESPName
                    end
                end
            end
        end
    end

    UpdateTracers()
end

AimlockTargetPlayer = "Nearest"

function getClosestPlayer()
    local myHrp = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
    if not myHrp then return nil end

    if AimlockTargetPlayer ~= "Nearest" and AimlockTargetPlayer ~= nil then
        local targetP = Players:FindFirstChild(AimlockTargetPlayer)
        if targetP and targetP.Character and targetP.Character:FindFirstChild("HumanoidRootPart") then
            local hum = targetP.Character:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health > 0 then return targetP.Character end
        end
    end

    local closest, closestDist = nil, math.huge
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= player and not BlacklistedPlayers[p.Name] and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health > 0 then
                local dist = (p.Character.HumanoidRootPart.Position - myHrp.Position).Magnitude
                if dist < closestDist and dist < maxRange then
                    closestDist = dist
                    closest = p.Character
                end
            end
        end
    end
    return closest
end

function getClosestNPC()
    local myHrp = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
    if not myHrp then return nil end
    local closest, closestDist = nil, math.huge
    local enemies = workspace:FindFirstChild("Enemies")
    if enemies then
        for _, npc in pairs(enemies:GetChildren()) do
            if npc:FindFirstChild("HumanoidRootPart") and npc:FindFirstChildOfClass("Humanoid") then
                if npc:FindFirstChildOfClass("Humanoid").Health > 0 then
                    local dist = (npc.HumanoidRootPart.Position - myHrp.Position).Magnitude
                    if dist < closestDist and dist < maxRange then
                        closestDist = dist
                        closest = npc
                    end
                end
            end
        end
    end
    return closest
end

local function addToggleElement(parent, labelText, defaultState, yPos, callback)
    local frame = Instance.new("Frame", parent)
    frame.Size = UDim2.new(1, -12, 0, 20)
    frame.Position = UDim2.new(0, 6, 0, yPos)
    frame.BackgroundTransparency = 1

    local label = Instance.new("TextLabel", frame)
    label.Text = labelText
    label.Font = Enum.Font.Gotham
    label.TextSize = 8.5
    label.TextColor3 = COLORS.TextWhite
    label.Size = UDim2.new(0.65, 0, 1, 0)
    label.BackgroundTransparency = 1
    label.TextXAlignment = Enum.TextXAlignment.Left

    local clickBtn = Instance.new("TextButton", frame)
    clickBtn.Size = UDim2.new(0, 30, 0, 13)
    clickBtn.Position = UDim2.new(1, -30, 0.5, -6)
    clickBtn.BackgroundColor3 = defaultState and currentThemeColor or COLORS.ToggleOff
    clickBtn.Text = defaultState and "ON" or "OFF"
    clickBtn.Font = Enum.Font.GothamBold
    clickBtn.TextSize = 7
    clickBtn.TextColor3 = defaultState and Color3.fromRGB(15, 10, 20) or COLORS.TextWhite
    Instance.new("UICorner", clickBtn).CornerRadius = UDim.new(0, 4)
    if defaultState then table.insert(themeFrames, clickBtn) end

    local state = defaultState
    local function refresh()
        if state then
            clickBtn.BackgroundColor3 = currentThemeColor
            clickBtn.Text = "ON"
            clickBtn.TextColor3 = Color3.fromRGB(15, 10, 20)
            if not table.find(themeFrames, clickBtn) then table.insert(themeFrames, clickBtn) end
        else
            clickBtn.BackgroundColor3 = COLORS.ToggleOff
            clickBtn.Text = "OFF"
            clickBtn.TextColor3 = COLORS.TextWhite
            local idx = table.find(themeFrames, clickBtn)
            if idx then table.remove(themeFrames, idx) end
        end
    end

    local function setExternalState(newState)
        state = newState
        refresh()
        callback(state)
    end

    table.insert(ToggleRegistry, function(val) setExternalState(val) end)

    clickBtn.MouseButton1Click:Connect(function()
        state = not state
        refresh()
        callback(state)
        if state then totalExecutions = totalExecutions + 1 end
    end)
    
    return setExternalState, clickBtn
end

local function formatNumber(n)
    if type(n) ~= "number" then return tostring(n) end
    local formatted = tostring(n)
    while true do
        local k
        formatted, k = string.gsub(formatted, "^(-?%d+)(%d%d%d)", '%1,%2')
        if k == 0 then break end
    end
    return formatted
end

local function getGameStat(statName)
    local val = nil
    local attr = player:GetAttribute(statName)
    if attr ~= nil then val = attr end

    if val == nil then
        local data = player:FindFirstChild("Data")
        if data then
            local lvlObj = data:FindFirstChild(statName)
            if lvlObj and lvlObj:IsA("ValueBase") then val = lvlObj.Value end
        end
    end

    if val == nil then
        local ls = player:FindFirstChild("leaderstats")
        if ls then
            local lvlObj = ls:FindFirstChild(statName)
            if lvlObj and lvlObj:IsA("ValueBase") then val = lvlObj.Value end
        end
    end
    return val
end

local function getLevelStat()
    local val = nil
    local attr = player:GetAttribute("Level")
    if attr ~= nil then val = attr end
    if val == nil then
        local data = player:FindFirstChild("Data")
        if data then
            local lvlObj = data:FindFirstChild("Level") or data:FindFirstChild("Nivel")
            if lvlObj and lvlObj:IsA("ValueBase") then val = lvlObj.Value end
        end
    end
    if val == nil then
        local ls = player:FindFirstChild("leaderstats")
        if ls then
            local lvlObj = ls:FindFirstChild("Level") or ls:FindFirstChild("Nivel")
            if lvlObj and lvlObj:IsA("ValueBase") then val = lvlObj.Value end
        end
    end
    return val
end

local function getBountyValue()
    local b = tonumber(getGameStat("Bounty")) or 0
    local h = tonumber(getGameStat("Honor")) or 0
    if b == 0 and h == 0 then
        pcall(function()
            local data = player:FindFirstChild("Data")
            if data then
                for _, child in pairs(data:GetChildren()) do
                    local nm = string.lower(child.Name)
                    if (string.find(nm, "bounty") or string.find(nm, "honor")) and child:IsA("ValueBase") then
                        local v = tonumber(child.Value) or 0
                        if v > b then b = v end
                    end
                end
            end
        end)
    end
    return math.max(b, h)
end

    currentThemeColor = THEMES[themeName]
    themeSelector.Text = themeName .. "  ▼"
    local isLight = isColorLight(currentThemeColor)

    for _, s in ipairs(themeStrokes) do if s and s.Parent then s.Color = currentThemeColor end end
    for _, t in ipairs(themeTexts) do if t and t.Parent then t.TextColor3 = currentThemeColor end end
    for _, f in ipairs(themeFrames) do 
        if f and f.Parent then 
            f.BackgroundColor3 = currentThemeColor 
            if f:IsA("TextButton") or f:IsA("TextLabel") then
                f.TextColor3 = isLight and Color3.fromRGB(15, 10, 20) or Color3.fromRGB(255, 255, 255)
            end
        end 
    end
    for _, drop in ipairs(rainContainer:GetChildren()) do
        if drop.Name == "RainDrop" then drop.BackgroundColor3 = currentThemeColor end
    end
    for _, child in ipairs(themeDropdown:GetChildren()) do
        if child:IsA("TextButton") and THEMES[child.Text] then child.TextColor3 = THEMES[child.Text] end
    end
    themeDropdown.Visible = false
end

for name, color in pairs(THEMES) do
    local opt = Instance.new("TextButton", themeDropdown)
    opt.Size = UDim2.new(1, 0, 0, 22)
    opt.BackgroundTransparency = 1
    opt.Text = name
    opt.Font = Enum.Font.GothamSemibold
    opt.TextSize = 8.5
    opt.TextColor3 = color
    opt.MouseButton1Click:Connect(function() applyNewTheme(name) end)
end

themeSelector.MouseButton1Click:Connect(function()
    themeDropdown.Visible = not themeDropdown.Visible
end)

UserInputService.InputBegan:Connect(function(input, gp)
    if not gp and input.KeyCode == Enum.KeyCode.F4 then
        mainFrame.Visible = not mainFrame.Visible
        openButton.Visible = not mainFrame.Visible
    end
end)
